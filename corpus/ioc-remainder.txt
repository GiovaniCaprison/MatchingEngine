# An immediate-or-cancel order takes what is there and its remainder leaves. The removal reports the
# quantity that never traded, and the order never appears as resting.

INSTRUMENT tick=5 lot=1 scale=4 min=1 max=1000000 band=500 open=100000 alloc=PRICE_TIME

SESSION    CONTINUOUS
STATE      CONTINUOUS

NEW        BUY  LIMIT GTC 100000 20
ACCEPTED   #1
RESTED     #1 BUY 100000 20

NEW        SELL LIMIT IOC 100000 50
ACCEPTED   #2
EXECUTED   @1 aggressor=#2 resting=#1 100000 20
REMOVED    #2 30 IMMEDIATE_OR_CANCEL_REMAINDER
