# FR-4.3 a replace names the order's whole quantity, not what is left of it
#
# A hundred with sixty traded, replaced to seventy. Seventy is the whole order, so ten is what
# should still be working. Read as a remainder it would have left seventy working and traded a
# hundred and thirty against an order for a hundred.

INSTRUMENT tick=5 lot=1 scale=4 min=1 max=1000000 band=500 open=100000 alloc=PRICE_TIME

SESSION    CONTINUOUS
STATE      CONTINUOUS

NEW        SELL LIMIT GTC 100000 100 p=2
ACCEPTED   #1
RESTED     #1 SELL 100000 100

NEW        BUY LIMIT GTC 100000 60 p=3
ACCEPTED   #2
EXECUTED   @1 aggressor=#2 resting=#1 100000 60

REPLACE    #1 70 100000
REDUCED    #1 10

NEW        BUY LIMIT GTC 100000 10 p=4
ACCEPTED   #3
EXECUTED   @2 aggressor=#3 resting=#1 100000 10
